cmake_minimum_required(VERSION 3.16)
project(NativeVoxel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
    message(FATAL_ERROR "This initial platform backend requires Windows (Win32/WGL/Winsock).")
endif()
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(NativeVoxel src/main.cpp src/platform.cpp src/renderer.cpp src/assets.cpp src/render_assets.cpp src/entity_models.cpp src/audio.cpp src/world.cpp src/world_io.cpp src/protocol.cpp src/game.cpp src/controller.cpp src/crafting.cpp src/gui_inventory.cpp src/picking.cpp src/settings.cpp src/gui_options.cpp src/skins.cpp src/first_person.cpp src/item_entities.cpp src/tile_entities.cpp src/local_session.cpp src/simulation.cpp src/simulation_entities.cpp src/simulation_containers.cpp src/brewing.cpp src/drops.cpp src/drop_entities.cpp src/worldgen.cpp src/worldgen_noise.cpp src/worldgen_biomes.cpp src/worldgen_surface.cpp src/worldgen_carvers.cpp src/worldgen_dimensions.cpp src/worldgen_features.cpp src/worldgen_trees.cpp src/worldgen_decoration.cpp src/worldgen_loot.cpp src/worldgen_dungeons.cpp src/worldgen_nether_population.cpp)
target_include_directories(NativeVoxel PRIVATE include)
target_sources(NativeVoxel PRIVATE src/entity_visuals.cpp src/particles.cpp src/resource_packs.cpp src/chat_text.cpp src/gui_hud.cpp src/server_list.cpp src/pickup.cpp src/projectiles.cpp)
target_compile_definitions(NativeVoxel PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
target_link_libraries(NativeVoxel PRIVATE OpenGL32 Gdi32 User32 Shell32 Winmm Ws2_32 Winhttp Dnsapi Ole32 Windowscodecs Uuid ZLIB::ZLIB OpenSSL::Crypto)
if(MSVC)
    target_compile_options(NativeVoxel PRIVATE /W4 /fp:strict /utf-8)
else()
    target_compile_options(NativeVoxel PRIVATE -Wall -Wextra -ffp-contract=off -fno-fast-math)
    if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(NativeVoxel PRIVATE stdc++fs)
    endif()
endif()
